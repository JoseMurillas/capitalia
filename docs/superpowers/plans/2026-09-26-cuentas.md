# Cuentas — Plan de implementación

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Que cada movimiento de Finanzas diga de qué cuenta salió, y que el saldo de cada cuenta —débito, ahorros, efectivo, billetera— sea la suma de lo que pasó por ella.

**Architecture:** Una tabla `Account` y una `AccountMovement` para lo que no es un movimiento de Finanzas (saldo inicial, transferencias, cajas, ajustes). `Transaction` gana `accountId` opcional. El saldo se calcula sumando las dos fuentes en vivo, nunca se guarda. Se sigue el patrón ya probado de `cash-boxes.ts`: cálculo puro → consultas → servicios transaccionales → acciones → RSC → cliente.

**Tech Stack:** Next.js 16 (App Router), Prisma 7, PostgreSQL, decimal.js, Zod 4, react-hook-form, shadcn/ui, Vitest, Playwright.

## Global Constraints

- **NO COMMITS.** No ejecutar `git add`, `git commit`, `git stash` ni nada que toque el índice o HEAD. Todo queda en el working tree.
- Comentarios de código en inglés; todo el texto que ve el usuario, en español.
- El dinero es `Decimal` en el servidor y `number` sólo en los DTO. Nunca aritmética con `number` sobre dinero: usar `toDecimal`, `sumMoney`, `toDbString`, `toNumber` de `@/lib/calculations`.
- Las fechas de negocio son `IsoDate` (`yyyy-MM-dd`) con `fromIsoDate` / `toIsoDate` / `todayIso`.
- `requireSession()` es la primera línea de toda consulta y toda acción. Las consultas llevan `import "server-only"`.
- Prisma 7 genera en `src/generated/prisma`. Tras tocar el esquema: `npx prisma migrate dev` y `npx prisma generate`.
- Next.js 16: `params` y `searchParams` son promesas; las páginas se tipan con `PageProps<"/ruta">`; `npx next typegen` antes de `npm run typecheck`.
- `revalidatePath` con ruta literal (`"/finanzas/cuentas"`); el prefijo de grupo `(dashboard)` sólo se usa en rutas dinámicas con `"page"`.
- El dev server corre en el puerto 3000 y PostgreSQL en el **5433**. Playwright: `E2E_PORT=3000`.
- No bloquear ninguna operación por saldo negativo (spec §4.1). La única validación de destino es que una transferencia no vaya a la misma cuenta.

- [ ] **Step 5: Implementar `src/lib/calculations/accounts.ts`**

```ts
import { type MoneyInput, sumMoney, toDecimal } from "./money";

export type AccountLedgerInput = {
  /** Signed movements: OPENING, TRANSFER, CASH_BOX, ADJUSTMENT. */
  movements: { amount: MoneyInput }[];
  /** Finance movements assigned to the account. */
  transactions: { type: "INCOME" | "EXPENSE"; amount: MoneyInput }[];
};

/**
 * What the account holds, read from its two sources at once. Nothing is stored,
 * so deleting a movement takes it out of the balance without anything to keep
 * in sync.
 */
export function accountBalance(input: AccountLedgerInput) {
  const fromMovements = sumMoney(input.movements.map((m) => toDecimal(m.amount)));
  const fromTransactions = sumMoney(
    input.transactions.map((t) => (t.type === "INCOME" ? toDecimal(t.amount) : toDecimal(t.amount).negated())),
  );
  return fromMovements.plus(fromTransactions);
}

/** What every account adds up to. */
export function accountsTotal(balances: MoneyInput[]) {
  return sumMoney(balances.map((balance) => toDecimal(balance)));
}
```

```ts
import { randomUUID } from "node:crypto";

import type Decimal from "decimal.js";

import type { Prisma } from "@/generated/prisma/client";
import { accountBalance, type MoneyInput, toDbString, toDecimal } from "@/lib/calculations";
import { fromIsoDate, type IsoDate } from "@/lib/dates";
import { prisma } from "@/lib/prisma";
import type {
  AccountAdjustmentInput,
  AccountInput,
  AccountTransferInput,
  CreateAccountInput,
} from "@/lib/validations/account";

import { NotFoundError, ServiceError } from "../errors";

export type Db = Prisma.TransactionClient | typeof prisma;

/** The account's balance, read inside whatever transaction the caller is running. */
export async function accountBalanceOf(db: Db, accountId: string): Promise<Decimal> {
  const [movements, transactions] = await Promise.all([
    db.accountMovement.findMany({ where: { accountId }, select: { amount: true } }),
    db.transaction.findMany({ where: { accountId }, select: { type: true, amount: true } }),
  ]);
  return accountBalance({ movements, transactions });
}

export async function assertAccountUsable(db: Db, accountId: string, field?: string) {
  const account = await db.account.findUnique({
    where: { id: accountId },
    select: { id: true, name: true, active: true },
  });
  if (!account) {
    const message = "La cuenta seleccionada no existe";
    throw new ServiceError(message, field ? { [field]: [message] } : undefined);
  }
  if (!account.active) {
    const message = `La cuenta ${account.name} está inactiva`;
    throw new ServiceError(message, field ? { [field]: ["La cuenta está inactiva"] } : undefined);
  }
  return { id: account.id, name: account.name };
}

type MovementDraft = {
  accountId: string;
  kind: "OPENING" | "TRANSFER" | "CASH_BOX" | "ADJUSTMENT";
  /** Signed. */
  amount: MoneyInput;
  movementDate: IsoDate;
  description: string;
  notes?: string | null;
  cashBoxMovementId?: string | null;
  transferGroupId?: string | null;
};

/** Every write to an account ledger goes through here, so nothing skips the shape. */
export async function writeAccountMovement(db: Db, draft: MovementDraft) {
  return db.accountMovement.create({
    data: {
      accountId: draft.accountId,
      kind: draft.kind,
      amount: toDbString(draft.amount),
      movementDate: fromIsoDate(draft.movementDate),
      description: draft.description,
      notes: draft.notes ?? null,
      cashBoxMovementId: draft.cashBoxMovementId ?? null,
      transferGroupId: draft.transferGroupId ?? null,
    },
  });
}

export async function createAccount(input: CreateAccountInput) {
  await prisma.$transaction(async (tx) => {
    const account = await tx.account.create({
      data: {
        name: input.name,
        kind: input.kind,
        issuer: input.issuer,
        last4: input.last4,
        notes: input.notes,
      },
    });
    // An opening balance is a movement like any other, so the balance stays a sum.
    if (toDecimal(input.openingBalance).gt(0)) {
      await writeAccountMovement(tx, {
        accountId: account.id,
        kind: "OPENING",
        amount: input.openingBalance,
        movementDate: input.openingDate,
        description: "Saldo inicial",
      });
    }
  });
}

export async function updateAccount(id: string, input: AccountInput) {
  const account = await prisma.account.findUnique({ where: { id }, select: { id: true } });
  if (!account) throw new NotFoundError("La cuenta");
  await prisma.account.update({
    where: { id },
    data: { name: input.name, kind: input.kind, issuer: input.issuer, last4: input.last4, notes: input.notes },
  });
}

export async function setAccountActive(id: string, active: boolean) {
  const account = await prisma.account.findUnique({ where: { id }, select: { id: true } });
  if (!account) throw new NotFoundError("La cuenta");
  await prisma.account.update({ where: { id }, data: { active } });
}

/**
 * Only an account nothing ever happened in can be removed. Its opening balance
 * does not count: it is the account being born, not something that happened.
 */
export async function deleteAccount(id: string) {
  const [account, movements, transactions] = await Promise.all([
    prisma.account.findUnique({ where: { id }, select: { id: true } }),
    prisma.accountMovement.count({ where: { accountId: id, kind: { not: "OPENING" } } }),
    prisma.transaction.count({ where: { accountId: id } }),
  ]);
  if (!account) throw new NotFoundError("La cuenta");
  if (movements > 0 || transactions > 0) {
    throw new ServiceError(
      "La cuenta ya tiene movimientos. Desactívala para dejar de usarla sin perder su historia.",
    );
  }
  await prisma.account.delete({ where: { id } });
}

export async function transferBetweenAccounts(fromId: string, input: AccountTransferInput) {
  if (fromId === input.toAccountId) {
    throw new ServiceError("Elige una cuenta distinta", { toAccountId: ["Elige una cuenta distinta"] });
  }
  await prisma.$transaction(async (tx) => {
    const from = await assertAccountUsable(tx, fromId);
    const to = await assertAccountUsable(tx, input.toAccountId, "toAccountId");
    // Both legs share the group, so the ledger can show them as one move.
    const transferGroupId = randomUUID();
    const description = input.description ?? `Transferencia a ${to.name}`;

    await writeAccountMovement(tx, {
      accountId: from.id,
      kind: "TRANSFER",
      amount: toDecimal(input.amount).negated(),
      movementDate: input.movementDate,
      description,
      notes: input.notes,
      transferGroupId,
    });
    await writeAccountMovement(tx, {
      accountId: to.id,
      kind: "TRANSFER",
      amount: input.amount,
      movementDate: input.movementDate,
      description: input.description ?? `Transferencia desde ${from.name}`,
      notes: input.notes,
      transferGroupId,
    });
  });
}

export async function adjustAccount(id: string, input: AccountAdjustmentInput) {
  await prisma.$transaction(async (tx) => {
    const account = await assertAccountUsable(tx, id);
    if (toDecimal(input.amount).isZero()) {
      throw new ServiceError("El ajuste no puede ser cero", { amount: ["El ajuste no puede ser cero"] });
    }
    await writeAccountMovement(tx, {
      accountId: account.id,
      kind: "ADJUSTMENT",
      amount: input.amount,
      movementDate: input.movementDate,
      description: input.description,
      notes: input.notes,
    });
  });
}

/** Puts an old movement into an account; this is the manual assignment flow. */
export async function assignTransactionAccount(transactionId: string, accountId: string) {
  const transaction = await prisma.transaction.findUnique({ where: { id: transactionId }, select: { id: true } });
  if (!transaction) throw new NotFoundError("El movimiento");
  await assertAccountUsable(prisma, accountId, "accountId");
  await prisma.transaction.update({ where: { id: transactionId }, data: { accountId } });
}
```

```ts
import "server-only";

import type { AccountKind } from "@/generated/prisma/enums";
import { accountBalance, accountsTotal, toNumber } from "@/lib/calculations";
import { prisma } from "@/lib/prisma";
import { requireSession } from "@/server/auth";

export type AccountDto = {
  id: string;
  name: string;
  kind: AccountKind;
  issuer: string | null;
  last4: string | null;
  active: boolean;
  notes: string | null;
  balance: number;
  /** Everything that ever touched it, to decide whether it can be deleted. */
  movementCount: number;
};

export type AccountsSummary = {
  total: number;
  /** Finance movements still without an account; while there are any, totals are partial. */
  unassignedCount: number;
  byKind: { kind: AccountKind; total: number; count: number }[];
};

/** Every account with its balance, active ones first and then by name. */
export async function listAccounts(): Promise<AccountDto[]> {
  await requireSession();
  const accounts = await prisma.account.findMany({
    orderBy: [{ active: "desc" }, { name: "asc" }],
    include: {
      movements: { select: { amount: true } },
      transactions: { select: { type: true, amount: true } },
    },
  });

  return accounts.map((account) => ({
    id: account.id,
    name: account.name,
    kind: account.kind,
    issuer: account.issuer,
    last4: account.last4,
    active: account.active,
    notes: account.notes,
    balance: toNumber(accountBalance({ movements: account.movements, transactions: account.transactions })),
    movementCount: account.movements.length + account.transactions.length,
  }));
}

export async function getAccountsSummary(): Promise<AccountsSummary> {
  await requireSession();
  const [accounts, unassignedCount] = await Promise.all([
    listAccounts(),
    prisma.transaction.count({ where: { accountId: null } }),
  ]);

  const active = accounts.filter((account) => account.active);
  const byKind = new Map<AccountKind, { total: number; count: number }>();
  for (const account of active) {
    const current = byKind.get(account.kind) ?? { total: 0, count: 0 };
    byKind.set(account.kind, { total: current.total + account.balance, count: current.count + 1 });
  }

  return {
    total: toNumber(accountsTotal(active.map((account) => account.balance))),
    unassignedCount,
    byKind: [...byKind.entries()].map(([kind, value]) => ({ kind, ...value })),
  };
}

/** What the selectors show: only accounts you can still use. */
export async function getAccountOptions() {
  await requireSession();
  return prisma.account.findMany({
    where: { active: true },
    orderBy: [{ name: "asc" }],
    select: { id: true, name: true, kind: true },
  });
}
```

- [ ] **Step 1: Añadir `getAccount` a `src/server/queries/accounts.ts`**

Trae los `AccountMovement` y las `Transaction` de la cuenta, los mezcla en una sola lista y calcula el saldo corriendo.

```ts
/**
 * The account's ledger, both sources in one list. The running balance follows
 * `createdAt`, not the business date: ordering a running balance by the date the
 * user typed makes an account open in the negative, which is what happened in
 * the cash box detail before it was fixed.
 */
export async function getAccount(id: string) {
  await requireSession();
  const account = await prisma.account.findUnique({
    where: { id },
    include: {
      movements: { orderBy: { createdAt: "asc" } },
      transactions: { orderBy: { createdAt: "asc" } },
    },
  });
  if (!account) return null;

  const entries = [
    ...account.movements.map((m) => ({
      id: m.id,
      source: "MOVEMENT" as const,
      kindKey: m.kind,
      description: m.description,
      date: toIsoDate(m.movementDate),
      amount: toNumber(m.amount),
      createdAt: m.createdAt,
      href: null as string | null,
    })),
    ...account.transactions.map((t) => ({
      id: t.id,
      source: "TRANSACTION" as const,
      kindKey: t.type,
      description: t.description,
      date: toIsoDate(t.transactionDate),
      amount: toNumber(t.type === "INCOME" ? toDecimal(t.amount) : toDecimal(t.amount).negated()),
      createdAt: t.createdAt,
      href: "/finanzas/movimientos",
    })),
  ].sort((a, b) => a.createdAt.getTime() - b.createdAt.getTime());

  let running = toDecimal(0);
  const withBalance = entries.map((entry) => {
    running = running.plus(toDecimal(entry.amount));
    return { ...entry, runningBalance: toNumber(running) };
  });

  return {
    id: account.id,
    name: account.name,
    kind: account.kind,
    issuer: account.issuer,
    last4: account.last4,
    active: account.active,
    notes: account.notes,
    balance: toNumber(running),
    movementCount: entries.length,
    entries: withBalance.reverse(),
  };
}
```
